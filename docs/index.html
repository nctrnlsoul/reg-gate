Reg-Gate watches the Federal Register for new U.S. rules on digital assets, sorts each one into four categories, and posts the result on the blockchain. Tokens check it before every transfer, and if updates stop for a week, transfers pause. Built on Chainlink CRE.

Gate a token on a category

Solidity
interface IRegGate {
    function statusOf(uint8 category) external view returns (uint8);
    function isAllowed(uint8 category, bool strict) external view returns (bool);
}

error TransferBlockedByRegGate(uint8 category, uint8 status);

function _transfer(address from, address to, uint256 value) private {
    if (to == address(0)) revert ZeroAddress();
    if (!gate.isAllowed(category, strict)) {
        revert TransferBlockedByRegGate(category, gate.statusOf(category));
    }
    // balance check and the move, as in any ERC-20
}
